# RSS / Social Media Feeds Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add social media feed subscriptions — poll RSS Bridge every 5 minutes, post new content as rich Discord embeds with image carousels and video fix-URL support.

**Architecture:** Extends the existing hexagonal architecture with new domain entities (`FeedSubscription`, `FeedPost`), two new ports (`FeedRepository`, `FeedFetcher`), four use cases, two new SQLAlchemy models in a separate `feed_models.py` (same `Base`), a background `FeedPoller` cog using `@tasks.loop`, and a `FeedCog` for slash commands.

**Tech Stack:** `aiohttp` (async HTTP), `feedparser` (RSS/Atom parsing), `discord.py tasks.loop` (background polling), SQLite (`seen_posts` deduplication table), alembic migration.

## Global Constraints

- Python ≥ 3.11 — use `X | None` union syntax, `list[str]` generics
- `pytest-asyncio` auto mode — **never** add `@pytest.mark.asyncio` to any test
- Application tests: real in-memory SQLite — no mocks for DB layer
- Infrastructure tests: real SQLite on `tmp_path` — no mocks for DB layer
- Bot tests: mock use cases only — no Discord connection
- Clean arch contract: `domain/` → stdlib only; `application/` → domain + stdlib; `infrastructure/` → domain + stdlib + sqlalchemy/aiosqlite/aiohttp/feedparser; `bot/` → application + stdlib + discord
- `feedparser.parse()` is synchronous — always pass already-fetched content string, never a URL
- All SQLAlchemy models use `Mapped` + `mapped_column` style (match `models.py`)
- `async_sessionmaker` used everywhere (match `repository.py`)
- All test fixtures match existing conftest patterns exactly — `session_factory` yields, `await engine.dispose()` in teardown

from abc import ABC, abstractmethod
from .feed_entities import FeedSubscription, FeedPost


class FeedRepository(ABC):
    @abstractmethod
    async def save(
        self, channel_id: str, platform: str, account_name: str, rss_url: str
    ) -> FeedSubscription | None: ...

    @abstractmethod
    async def delete(self, feed_id: int, channel_id: str) -> bool: ...

    @abstractmethod
    async def list_for_channel(self, channel_id: str) -> list[FeedSubscription]: ...

    @abstractmethod
    async def list_all(self) -> list[FeedSubscription]: ...

    @abstractmethod
    async def is_seen(self, feed_id: int, guid: str) -> bool: ...

    @abstractmethod
    async def mark_seen(self, feed_id: int, guid: str) -> None: ...

    @abstractmethod
    async def prune_seen(self, older_than_days: int) -> None: ...


class FeedFetcher(ABC):
    @abstractmethod
    async def fetch(self, url: str) -> list[FeedPost]: ...

from datetime import datetime

from sqlalchemy import Integer, String, DateTime, ForeignKey, UniqueConstraint, Index
from sqlalchemy.orm import Mapped, mapped_column, relationship

from .models import Base


class FeedSubscriptionModel(Base):
    __tablename__ = "feed_subscriptions"
    __table_args__ = (
        UniqueConstraint("channel_id", "rss_url", name="uq_channel_rss_url"),
    )

    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    channel_id: Mapped[str] = mapped_column(String, nullable=False)
    platform: Mapped[str] = mapped_column(String, nullable=False)
    account_name: Mapped[str] = mapped_column(String, nullable=False)
    rss_url: Mapped[str] = mapped_column(String, nullable=False)
    created_at: Mapped[datetime] = mapped_column(DateTime, nullable=False)

    seen_posts: Mapped[list["SeenPostModel"]] = relationship(
        back_populates="subscription", cascade="all, delete-orphan"
    )


class SeenPostModel(Base):
    __tablename__ = "seen_posts"
    __table_args__ = (
        Index("ix_seen_posts_feed_guid", "feed_id", "guid"),
    )

    id: Mapped[int] = mapped_column(Integer, primary_key=True, autoincrement=True)
    feed_id: Mapped[int] = mapped_column(
        Integer, ForeignKey("feed_subscriptions.id", ondelete="CASCADE"), nullable=False
    )
    guid: Mapped[str] = mapped_column(String, nullable=False)
    seen_at: Mapped[datetime] = mapped_column(DateTime, nullable=False)

    subscription: Mapped["FeedSubscriptionModel"] = relationship(back_populates="seen_posts")
```

# lilwman/infrastructure/db/feed_repository.py
from datetime import datetime, timezone, timedelta

from sqlalchemy import select, delete
from sqlalchemy.exc import IntegrityError

from lilwman.domain.feed_entities import FeedSubscription
from lilwman.domain.feed_ports import FeedRepository
from .feed_models import FeedSubscriptionModel, SeenPostModel


class SQLAlchemyFeedRepository(FeedRepository):
    def __init__(self, session_factory) -> None:
        self._session_factory = session_factory

    async def save(
        self, channel_id: str, platform: str, account_name: str, rss_url: str
    ) -> FeedSubscription | None:
        now = datetime.now(timezone.utc)
        model = FeedSubscriptionModel(
            channel_id=channel_id,
            platform=platform,
            account_name=account_name,
            rss_url=rss_url,
            created_at=now,
        )
        async with self._session_factory() as session:
            async with session.begin():
                session.add(model)
                try:
                    await session.flush()
                except IntegrityError:
                    return None
        return FeedSubscription(
            id=model.id,
            channel_id=channel_id,
            platform=platform,
            account_name=account_name,
            rss_url=rss_url,
            created_at=now,
        )

    async def delete(self, feed_id: int, channel_id: str) -> bool:
        async with self._session_factory() as session:
            async with session.begin():
                result = await session.execute(
                    select(FeedSubscriptionModel).where(
                        FeedSubscriptionModel.id == feed_id,
                        FeedSubscriptionModel.channel_id == channel_id,
                    )
                )
                model = result.scalar_one_or_none()
                if model is None:
                    return False
                await session.delete(model)
                return True

    async def list_for_channel(self, channel_id: str) -> list[FeedSubscription]:
        async with self._session_factory() as session:
            result = await session.execute(
                select(FeedSubscriptionModel)
                .where(FeedSubscriptionModel.channel_id == channel_id)
                .order_by(FeedSubscriptionModel.created_at.asc())
            )
            return [self._to_entity(m) for m in result.scalars().all()]

    async def list_all(self) -> list[FeedSubscription]:
        async with self._session_factory() as session:
            result = await session.execute(
                select(FeedSubscriptionModel).order_by(FeedSubscriptionModel.id.asc())
            )
            return [self._to_entity(m) for m in result.scalars().all()]

    async def is_seen(self, feed_id: int, guid: str) -> bool:
        async with self._session_factory() as session:
            result = await session.execute(
                select(SeenPostModel).where(
                    SeenPostModel.feed_id == feed_id,
                    SeenPostModel.guid == guid,
                )
            )
            return result.scalar_one_or_none() is not None

    async def mark_seen(self, feed_id: int, guid: str) -> None:
        now = datetime.now(timezone.utc)
        async with self._session_factory() as session:
            async with session.begin():
                session.add(SeenPostModel(feed_id=feed_id, guid=guid, seen_at=now))

    async def prune_seen(self, older_than_days: int) -> None:
        cutoff = datetime.now(timezone.utc) - timedelta(days=older_than_days)
        async with self._session_factory() as session:
            async with session.begin():
                await session.execute(
                    delete(SeenPostModel).where(SeenPostModel.seen_at < cutoff)
                )

    def _to_entity(self, model: FeedSubscriptionModel) -> FeedSubscription:
        return FeedSubscription(
            id=model.id,
            channel_id=model.channel_id,
            platform=model.platform,
            account_name=model.account_name,
            rss_url=model.rss_url,
            created_at=model.created_at,
        )

# lilwman/infrastructure/http/feed_fetcher.py
import re
import calendar
import aiohttp
import feedparser

from datetime import datetime, timezone

from lilwman.domain.feed_entities import FeedPost
from lilwman.domain.feed_ports import FeedFetcher

_TIMEOUT = aiohttp.ClientTimeout(total=15)


class AiohttpFeedFetcher(FeedFetcher):
    async def fetch(self, url: str) -> list[FeedPost]:
        async with aiohttp.ClientSession(timeout=_TIMEOUT) as session:
            async with session.get(url) as response:
                response.raise_for_status()
                content = await response.text()
        return _parse_feed(content)


def _parse_feed(content: str) -> list[FeedPost]:
    parsed = feedparser.parse(content)
    posts = []
    for entry in parsed.entries:
        guid = entry.get("id") or entry.get("link", "")
        title = entry.get("title", "")[:256]
        url = entry.get("link", "")
        description = _strip_html(entry.get("summary", ""))[:300]
        author = entry.get("author", "")
        published_at = _parse_date(entry)
        images = _extract_images(entry)
        video_url = _extract_video(entry)
        posts.append(FeedPost(
            guid=guid,
            title=title,
            url=url,
            description=description,
            author=author,
            published_at=published_at,
            images=images,
            video_url=video_url,
        ))
    return posts


def _parse_date(entry) -> datetime:
    if getattr(entry, "published_parsed", None):
        return datetime.fromtimestamp(
            calendar.timegm(entry.published_parsed), tz=timezone.utc
        )
    return datetime.now(timezone.utc)


def _strip_html(text: str) -> str:
    return re.sub(r"<[^>]+>", "", text).strip()


def _extract_images(entry) -> list[str]:
    images = []
    for media in getattr(entry, "media_content", []):
        if media.get("medium") == "image" and media.get("url"):
            images.append(media["url"])
    if not images:
        for enc in getattr(entry, "enclosures", []):
            if enc.get("type", "").startswith("image/") and enc.get("href"):
                images.append(enc["href"])
    return images


def _extract_video(entry) -> str | None:
    for media in getattr(entry, "media_content", []):
        if media.get("medium") == "video" and media.get("url"):
            return media["url"]
    for enc in getattr(entry, "enclosures", []):
        if enc.get("type", "").startswith("video/") and enc.get("href"):
            return enc["href"]
    return None
```

from lilwman.domain.feed_entities import FeedSubscription
from lilwman.domain.feed_ports import FeedRepository

_BRIDGES: dict[str, str] = {
    "instagram":  "Instagram&username={account}",
    "tiktok":     "TikTok&username={account}",
    "twitter":    "Twitter&u={account}",
    "reddit":     "Reddit&r={account}",
    "youtube":    "YouTube&channel={account}",
    "twitch":     "Twitch&channel={account}",
    "hackernews": "HackerNews&category={account}",
}

SUPPORTED_PLATFORMS: set[str] = set(_BRIDGES.keys())


def build_rss_url(rss_bridge_host: str, platform: str, account: str) -> str:
    bridge_params = _BRIDGES[platform].format(account=account)
    return f"{rss_bridge_host}/?action=display&bridge={bridge_params}&format=Atom"


class AddFeedUseCase:
    def __init__(self, repo: FeedRepository, rss_bridge_url: str) -> None:
        self._repo = repo
        self._rss_bridge_url = rss_bridge_url

    async def execute(
        self, channel_id: str, platform: str, account_name: str
    ) -> FeedSubscription | None:
        rss_url = build_rss_url(self._rss_bridge_url, platform, account_name)
        return await self._repo.save(channel_id, platform, account_name, rss_url)

    async def execute_custom_url(
        self, channel_id: str, rss_url: str
    ) -> FeedSubscription | None:
        account_name = urlparse(rss_url).netloc or rss_url
        return await self._repo.save(channel_id, "custom", account_name, rss_url)

# lilwman/application/poll_feeds.py
from lilwman.domain.feed_entities import FeedSubscription, FeedPost
from lilwman.domain.feed_ports import FeedRepository, FeedFetcher


class PollFeedsUseCase:
    def __init__(self, repo: FeedRepository, fetcher: FeedFetcher) -> None:
        self._repo = repo
        self._fetcher = fetcher

    async def execute(self) -> list[tuple[FeedSubscription, list[FeedPost]]]:
        results = []
        subscriptions = await self._repo.list_all()
        for sub in subscriptions:
            try:
                posts = await self._fetcher.fetch(sub.rss_url)
            except Exception:
                continue
            new_posts = []
            for post in posts:
                if not await self._repo.is_seen(sub.id, post.guid):
                    new_posts.append(post)
            new_posts.sort(key=lambda p: p.published_at)
            new_posts = new_posts[:5]
            for post in new_posts:
                await self._repo.mark_seen(sub.id, post.guid)
            if new_posts:
                results.append((sub, new_posts))
        await self._repo.prune_seen(older_than_days=30)
        return results
```

# lilwman/bot/feed_paginator.py
import discord

from lilwman.bot.feed_utils import get_platform_colour
from lilwman.domain.feed_entities import FeedPost, FeedSubscription


class FeedImagePaginator(discord.ui.View):
    def __init__(self, post: FeedPost, subscription: FeedSubscription) -> None:
        super().__init__(timeout=120)
        self.post = post
        self.subscription = subscription
        self.page = 0

    @property
    def total_pages(self) -> int:
        return len(self.post.images)

    def make_embed(self, colour: int) -> discord.Embed:
        embed = discord.Embed(
            title=self.post.title,
            url=self.post.url,
            description=self.post.description,
            color=colour,
        )
        embed.set_author(
            name=f"@{self.subscription.account_name} on {self.subscription.platform}"
        )
        embed.set_image(url=self.post.images[self.page])
        embed.set_footer(text=f"Image {self.page + 1}/{self.total_pages}")
        embed.timestamp = self.post.published_at
        return embed

    @discord.ui.button(label="◀", style=discord.ButtonStyle.secondary)
    async def prev(self, interaction: discord.Interaction, button: discord.ui.Button) -> None:
        self.page = max(0, self.page - 1)
        await interaction.response.edit_message(
            embed=self.make_embed(get_platform_colour(self.subscription.platform)), view=self
        )

    @discord.ui.button(label="▶", style=discord.ButtonStyle.secondary)
    async def next(self, interaction: discord.Interaction, button: discord.ui.Button) -> None:
        self.page = min(self.total_pages - 1, self.page + 1)
        await interaction.response.edit_message(
            embed=self.make_embed(get_platform_colour(self.subscription.platform)), view=self
        )

# lilwman/bot/commands/feed.py
import discord
from discord import app_commands
from discord.ext import commands

from lilwman.application.add_feed import SUPPORTED_PLATFORMS


class FeedCog(commands.Cog):
    def __init__(self, bot) -> None:
        self.bot = bot

    feed = app_commands.Group(name="feed", description="Manage social media feed subscriptions")

    @feed.command(name="add", description="Subscribe this channel to a social media account")
    @app_commands.describe(
        platform=f"Platform ({', '.join(sorted(SUPPORTED_PLATFORMS))})",
        account="Account name, username, or subreddit",
    )
    async def feed_add(
        self, interaction: discord.Interaction, platform: str, account: str
    ) -> None:
        if not interaction.user.guild_permissions.administrator:
            await interaction.response.send_message(
                "Administrator permission required.", ephemeral=True
            )
            return
        platform = platform.lower().strip()
        if platform not in SUPPORTED_PLATFORMS:
            await interaction.response.send_message(
                f"Unknown platform **{platform}**. "
                f"Supported: {', '.join(sorted(SUPPORTED_PLATFORMS))}. "
                "For a custom RSS URL use `/feed addurl`.",
                ephemeral=True,
            )
            return
        channel_id = str(interaction.channel_id)
        sub = await self.bot.add_feed.execute(channel_id, platform, account)
        if sub is None:
            await interaction.response.send_message(
                f"Already subscribed to **{account}** on **{platform}** in this channel.",
                ephemeral=True,
            )
            return
        await interaction.response.send_message(
            f"Subscribed to **@{account}** on **{platform}** (id `{sub.id}`).",
            ephemeral=True,
        )

    @feed.command(name="addurl", description="Subscribe this channel to a custom RSS/Atom URL")
    @app_commands.describe(url="Full RSS or Atom feed URL")
    async def feed_addurl(self, interaction: discord.Interaction, url: str) -> None:
        if not interaction.user.guild_permissions.administrator:
            await interaction.response.send_message(
                "Administrator permission required.", ephemeral=True
            )
            return
        channel_id = str(interaction.channel_id)
        sub = await self.bot.add_feed.execute_custom_url(channel_id, url)
        if sub is None:
            await interaction.response.send_message(
                "Already subscribed to that URL in this channel.", ephemeral=True
            )
            return
        await interaction.response.send_message(
            f"Subscribed to custom feed (id `{sub.id}`).", ephemeral=True
        )

    @feed.command(name="remove", description="Remove a feed subscription from this channel")
    @app_commands.describe(id="Feed subscription id (from /feed list)")
    async def feed_remove(self, interaction: discord.Interaction, id: int) -> None:
        if not interaction.user.guild_permissions.administrator:
            await interaction.response.send_message(
                "Administrator permission required.", ephemeral=True
            )
            return
        channel_id = str(interaction.channel_id)
        deleted = await self.bot.remove_feed.execute(id, channel_id)
        if deleted:
            await interaction.response.send_message(f"Removed feed **#{id}**.", ephemeral=True)
        else:
            await interaction.response.send_message(
                f"Feed **#{id}** not found in this channel.", ephemeral=True
            )

    @feed.command(name="list", description="List feed subscriptions for this channel")

                value=f"@{sub.account_name}",
                inline=False,
            )
        await interaction.response.send_message(embed=embed, ephemeral=True)

# lilwman/bot/poller.py
import logging

import discord
from discord.ext import commands, tasks

from lilwman.application.poll_feeds import PollFeedsUseCase
from lilwman.bot.feed_paginator import FeedImagePaginator
from lilwman.bot.feed_utils import get_fix_url, get_platform_colour
from lilwman.domain.feed_entities import FeedPost, FeedSubscription

log = logging.getLogger(__name__)


class FeedPoller(commands.Cog):
    def __init__(self, bot, poll_feeds: PollFeedsUseCase) -> None:
        self.bot = bot
        self.poll_feeds = poll_feeds

    async def cog_load(self) -> None:
        self._poll_loop.start()

    def cog_unload(self) -> None:
        self._poll_loop.cancel()

    @tasks.loop(minutes=5)
    async def _poll_loop(self) -> None:
        await self._do_poll()

    @_poll_loop.before_loop
    async def _before_poll(self) -> None:
        await self.bot.wait_until_ready()

    async def _do_poll(self) -> None:
        try:
            results = await self.poll_feeds.execute()
        except Exception:
            log.exception("Feed poll error")
            return
        for sub, posts in results:
            for post in posts:
                await self._post(sub, post)

    async def _post(self, sub: FeedSubscription, post: FeedPost) -> None:
        channel = self.bot.get_channel(int(sub.channel_id))
        if channel is None:
            log.warning("Channel %s not found for feed %s", sub.channel_id, sub.id)
            return
        try:
            colour = get_platform_colour(sub.platform)
            if post.video_url:
                fix = get_fix_url(sub.platform, post.url)
                if fix:
                    await channel.send(fix)
                    return
            if len(post.images) > 1:
                paginator = FeedImagePaginator(post, sub)
                await channel.send(embed=paginator.make_embed(colour), view=paginator)
                return
            if post.images:
                embed = _make_post_embed(post, sub, colour)
                embed.set_image(url=post.images[0])
                await channel.send(embed=embed)
                return
            await channel.send(embed=_make_post_embed(post, sub, colour))
        except Exception:
            log.warning("Failed to post to channel %s", sub.channel_id, exc_info=True)


def _make_post_embed(
    post: FeedPost, sub: FeedSubscription, colour: int
) -> discord.Embed:
    embed = discord.Embed(
        title=post.title,
        url=post.url,
        description=post.description,
        color=colour,
    )
    embed.set_author(name=f"@{sub.account_name} on {sub.platform}")
    embed.timestamp = post.published_at
    return embed
```
